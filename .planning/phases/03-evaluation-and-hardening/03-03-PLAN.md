```python
def display_delta_table(metrics, title):
    lines = [f"### {title}", "", "| 指标 | 微调前 | 微调后 | 变化 | 变化率 |", "|------|--------|--------|------|--------|"]
    for key, val in metrics.items():
        if isinstance(val, dict) and "before" in val:
            sign = "+" if val["delta"] >= 0 else ""
            lines.append(f"| {key} | {val['before']:.4f} | {val['after']:.4f} | {sign}{val['delta']:.4f} | {sign}{val['delta_pct']:.1f}% |")
    display(Markdown("\n".join(lines)))
```
